cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(p4wifibench)

if(CONFIG_P4W_SCAN_ONLY)
  target_sources(app PRIVATE src/scan.c)
elseif(CONFIG_P4W_NETWORK_ONLY)
  target_sources(app PRIVATE src/network.c)
  if(CONFIG_P4W_NETWORK_TLS)
    target_sources(app PRIVATE src/tls.c)
  endif()
else()
  target_sources(app PRIVATE src/main.c)
endif()
