cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(tessera_core)

if(TEST_WDT_HARDWARE)
  target_sources(app PRIVATE src/wdt_hardware.c)
  if(TEST_WDT_HW_NOT_READY)
    target_compile_definitions(app PRIVATE TEST_WDT_HW_NOT_READY)
  endif()
elseif(TEST_WDT_START_FAILURES)
  target_sources(app PRIVATE src/wdt_start.c)
  target_link_options(app PUBLIC -Wl,--wrap=task_wdt_init -Wl,--wrap=task_wdt_add)
  if(TEST_WDT_REQUIRED_MISSING)
    target_compile_definitions(app PRIVATE TEST_WDT_REQUIRED_MISSING)
  endif()
else()
  target_sources(app PRIVATE src/main.c)
endif()
