cmake_minimum_required(VERSION 3.19)
project(Entelea VERSION 0.1.0 LANGUAGES C)

include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_architecture.py"
  RESULT_VARIABLE policy_result
)
if(NOT policy_result EQUAL 0)
  message(FATAL_ERROR "Source classification / R01 architecture check failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/build-support/source-policy.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/build-support/source-policy.json" source_policy)

function(policy_sources out group)
  string(JSON count LENGTH "${source_policy}" ${group})
  set(paths "")
  if(count GREATER 0)
    math(EXPR last "${count} - 1")
    foreach(index RANGE 0 ${last})
      string(JSON path GET "${source_policy}" ${group} ${index})
      list(APPEND paths "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
    endforeach()
  endif()
  set(${out} "${paths}" PARENT_SCOPE)
endfunction()

policy_sources(common_sources common)
add_library(entelea_common STATIC ${common_sources})
target_include_directories(entelea_common PUBLIC framework/include)
target_compile_features(entelea_common PUBLIC c_std_11)
set_target_properties(entelea_common PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
policy_sources(compat_sources compat)
add_library(entelea_compat STATIC EXCLUDE_FROM_ALL ${compat_sources})
target_link_libraries(entelea_compat PUBLIC entelea_common)
target_include_directories(entelea_compat PUBLIC framework/compat/include)
set_target_properties(entelea_compat PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(entelea_common PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  target_compile_options(entelea_compat PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
endif()
option(ENTELEA_SANITIZERS "Run host tests with address/undefined sanitizers" OFF)
if(ENTELEA_SANITIZERS)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require a supported GNU/Clang host compiler")
  endif()
  target_compile_options(entelea_common PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(entelea_common PUBLIC -fsanitize=address,undefined)
endif()

# Diagnostics are exact independent targets and are never linked into common.
string(JSON diagnostic_count LENGTH "${source_policy}" diagnostics)
if(diagnostic_count GREATER 0)
  math(EXPR diagnostic_last "${diagnostic_count} - 1")
  foreach(index RANGE 0 ${diagnostic_last})
    string(JSON name MEMBER "${source_policy}" diagnostics ${index})
    policy_sources(paths "diagnostics;${name}")
    add_executable(diagnostic_${name} EXCLUDE_FROM_ALL ${paths})
  endforeach()
endif()

if(BUILD_TESTING)
  policy_sources(test_sources contract_tests)
  add_executable(entelea_contract_tests ${test_sources})
  target_link_libraries(entelea_contract_tests PRIVATE entelea_compat)
  set_target_properties(entelea_contract_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(entelea_contract_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME common_contracts COMMAND entelea_contract_tests)
  add_test(NAME architecture COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_architecture.py")
  add_test(NAME migration_sources COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_migration.py" --sources)
  add_test(NAME governance_tests COMMAND "${Python3_EXECUTABLE}" -m unittest discover
    -s "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests" -v)
  add_test(NAME dependency_manifest COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_dependencies.py")
  add_test(NAME legacy_equivalence COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_legacy_equivalence.py" --cc "${CMAKE_C_COMPILER}")
endif()
